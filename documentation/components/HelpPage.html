import {
  helpCircleOutline,
  shieldCheckmarkOutline,
  personOutline,
  documentTextOutline,
  keyOutline,
  accessibilityOutline,
  layersOutline,
  logOutOutline,

import { NotificationService } from '../../core/services/notification.service';

/**
 * Manual de Usuario Embebido y Sistema de Ayuda Contextual (RA 6 & FASE 5).
 * Proporciona documentación interactiva integrada en la propia PWA.
 */
@Component({
  selector: 'app-help',
  templateUrl: './help.page.html',
  styleUrls: ['./help.page.scss'],
  changeDetection: ChangeDetectionStrategy.OnPush,
  standalone: true,
  imports: [
    CommonModule,
    IonHeader,
    IonToolbar,
    IonTitle,
    IonButtons,
    IonMenuButton,
    IonContent,
    IonCard,
    IonCardHeader,
    IonCardTitle,
    IonCardSubtitle,
    IonCardContent,
    IonAccordionGroup,
    IonAccordion,
    IonItem,
    IonLabel,
    IonIcon,
    IonBadge,
    IonButton
  ]
})
export class HelpPage {
  public authService = inject(AuthService);
  private notificationService = inject(NotificationService);
  private router = inject(Router);

  constructor() {
    addIcons({
      helpCircleOutline,
      shieldCheckmarkOutline,
      personOutline,
      documentTextOutline,
      keyOutline,
      accessibilityOutline,
      layersOutline,
      logOutOutline,
      logInOutline
    });
  }

  public logout(): void {
    this.authService.logout();
    this.notificationService.showToast('Sesión cerrada correctamente.', 'primary');
    this.router.navigate(['/login']);
  }

  public goToLogin(): void {
    this.router.navigate(['/login']);
  }
}
