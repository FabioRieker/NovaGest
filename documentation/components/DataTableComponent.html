import {
  Component,
  Input,
  Output,
  EventEmitter,
  ChangeDetectionStrategy,
  OnInit,
  OnChanges,
  OnDestroy,
  SimpleChanges,
  signal
} from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import {
  IonSearchbar,
  IonButton,
  IonIcon,
  IonSpinner,
  IonBadge
} from '@ionic/angular';
import { addIcons } from 'ionicons';
import {
  arrowUpOutline,
  arrowDownOutline,
  chevronBackOutline,
  chevronForwardOutline,
  createOutline,
  trashOutline,
  eyeOutline,
  swapVerticalOutline,
  imageOutline,
  closeOutline
} from 'ionicons/icons';
import { Subject, Subscription } from 'rxjs';
import { debounceTime, distinctUntilChanged } from 'rxjs/operators';

export interface TableColumn {
  key: string;
  label: string;
  sortable?: boolean;
  type?: 'text' | 'currency' | 'badge' | 'date' | 'image';
  width?: string;
}

export interface TableActionPayload<T = any> {
  action: 'view' | 'edit' | 'delete';
  item: T;
}

/**
 * Componente Visual Personalizado 1: Tabla de Datos Avanzada Reutilizable (RA 3).
 * Completamente agnóstico de la vista contenedora, configurable mediante @Input()
 * y emisor de interacciones mediante @Output().
 * Cumple con ChangeDetectionStrategy.OnPush y normativas WCAG de accesibilidad.
 */
@Component({
  selector: 'app-data-table',
  templateUrl: './data-table.component.html',
  styleUrls: ['./data-table.component.scss'],
  changeDetection: ChangeDetectionStrategy.OnPush,
  standalone: true,
  imports: [
    CommonModule,
    FormsModule,
    IonSearchbar,
    IonButton,
    IonIcon,
    IonSpinner,
    IonBadge
  ]
})
export class DataTableComponent implements OnInit, OnChanges, OnDestroy {
  @Input() columns: TableColumn[] = [];
  @Input() data: any[] = [];
  @Input() pageSize = 6;
  @Input() searchPlaceholder = 'Buscar registros...';
  @Input() loading = false;
  @Input() emptyMessage = 'No se encontraron registros.';
  @Input() showActions = true;
  @Input() canEdit = true;
  @Input() canDelete = true;
  @Input() canView = true;

  @Output() rowClick = new EventEmitter<any>();
  @Output() actionClick = new EventEmitter<TableActionPayload>();
  @Output() sortChange = new EventEmitter<{ column: string; direction: 'asc' | 'desc' }>();
  @Output() pageChange = new EventEmitter<number>();

  // Estado interno de la tabla
  public searchTerm = '';
  public currentPage = 1;
  public sortColumn = '';
  public sortDirection: 'asc' | 'desc' = 'asc';

  public filteredData: any[] = [];
  public paginatedData: any[] = [];
  public totalPages = 1;

  private searchSubject = new Subject<string>();
  private subscriptions = new Subscription();

  public previewModal = signal<{ url: string; title: string } | null>(null);

  constructor() {
    addIcons({
      arrowUpOutline,
      arrowDownOutline,
      chevronBackOutline,
      chevronForwardOutline,
      createOutline,
      trashOutline,
      eyeOutline,
      swapVerticalOutline,
      imageOutline,
      closeOutline
    });
  }

  ngOnInit(): void {
    // Configurar debounce en la barra de búsqueda para evitar recomputaciones innecesarias
    this.subscriptions.add(
      this.searchSubject.pipe(
        debounceTime(250),
        distinctUntilChanged()
      ).subscribe(term => {
        this.searchTerm = term;
        this.currentPage = 1;
        this.applyFilterAndPagination();
      })
    );

    this.applyFilterAndPagination();
  }

  ngOnChanges(changes: SimpleChanges): void {
    if (changes['data'] || changes['pageSize']) {
      this.applyFilterAndPagination();
    }
  }

  ngOnDestroy(): void {
    // Evitar memory leaks liberando suscripciones de Observables
    this.subscriptions.unsubscribe();
  }

  public onSearchInput(event: any): void {
    const val = event.target?.value ?? '';
    this.searchSubject.next(val);
  }

  public onSort(col: TableColumn): void {
    if (!col.sortable) return;

    if (this.sortColumn === col.key) {
      this.sortDirection = this.sortDirection === 'asc' ? 'desc' : 'asc';
    } else {
      this.sortColumn = col.key;
      this.sortDirection = 'asc';
    }

    this.sortChange.emit({ column: this.sortColumn, direction: this.sortDirection });
    this.applyFilterAndPagination();
  }

  public setPage(page: number): void {
    if (page < 1 || page > this.totalPages || page === this.currentPage) return;
    this.currentPage = page;
    this.pageChange.emit(this.currentPage);
    this.applyPaginationOnly();
  }

  public onRowSelect(item: any): void {
    this.rowClick.emit(item);
  }

  public triggerAction(action: 'view' | 'edit' | 'delete', item: any, event: Event): void {
    event.stopPropagation();
    this.actionClick.emit({ action, item });
  }

  private applyFilterAndPagination(): void {
    let result = [...(this.data || [])];

    // Búsqueda en todos los campos visibles
    if (this.searchTerm.trim()) {
      const q = this.searchTerm.toLowerCase().trim();
      result = result.filter(item =>
        this.columns.some(col => {
          const val = item[col.key];
          return val !== null && val !== undefined && String(val).toLowerCase().includes(q);
        })
      );
    }

    // Ordenación
    if (this.sortColumn) {
      const dir = this.sortDirection === 'desc' ? -1 : 1;
      result.sort((a, b) => {
        const valA = a[this.sortColumn];
        const valB = b[this.sortColumn];
        if (typeof valA === 'number' && typeof valB === 'number') {
          return dir * (valA - valB);
        }
        return dir * String(valA || '').localeCompare(String(valB || ''));
      });
    }

    this.filteredData = result;
    this.totalPages = Math.max(1, Math.ceil(this.filteredData.length / this.pageSize));

    if (this.currentPage > this.totalPages) {
      this.currentPage = 1;
    }

    this.applyPaginationOnly();
  }

  private applyPaginationOnly(): void {
    const start = (this.currentPage - 1) * this.pageSize;
    const end = start + this.pageSize;
    this.paginatedData = this.filteredData.slice(start, end);
  }

  public getBadgeColor(status: string): string {
    switch (status) {
      case 'in_stock': return 'success';
      case 'low_stock': return 'warning';
      case 'out_of_stock': return 'danger';
      case 'admin': return 'primary';
      case 'user': return 'medium';
      default: return 'secondary';
    }
  }

  public formatCellValue(item: any, col: TableColumn): string {
    const val = item[col.key];
    if (val === null || val === undefined) return '-';
    if (col.type === 'currency') {
      return `${Number(val).toFixed(2)} €`;
    }
    if (col.type === 'date') {
      try {
        return new Date(val).toLocaleDateString('es-ES', { day: '2-digit', month: 'short', year: 'numeric' });
      } catch {
        return String(val);
      }
    }
    return String(val);
  }

  public openImagePreview(url: string, title: string, event: Event): void {
    event.stopPropagation();
    if (url) {
      this.previewModal.set({ url, title });
    }
  }

  public closeImagePreview(): void {
    this.previewModal.set(null);
  }
}

  <div class="table-toolbar">
    <div class="search-wrapper">
      <ion-searchbar
        [placeholder]="searchPlaceholder"
        [value]="searchTerm"
        (ionInput)="onSearchInput($event)"
        debounce="0"

      ></ion-searchbar>
    </div>

  <!-- Estado de Carga -->
  @if (loading) {
    <div class="table-loading-state" role="status" aria-live="polite">

      <span>Cargando datos del servidor...</span>

  <!-- Contenedor Responsive de la Tabla -->
  <div class="table-responsive-wrapper">
    <table class="custom-table" role="table">
      <caption class="sr-only">Listado de registros del sistema</caption>
      <thead>
        <tr role="row">
          @for (col of columns; track col.key) {
            <th
              role="columnheader"
              scope="col"
              [style.width]="col.width || 'auto'"
              [attr.aria-sort]="sortColumn === col.key ? (sortDirection === 'asc' ? 'ascending' : 'descending') : 'none'"
              [class.sortable]="col.sortable"
              (click)="onSort(col)"
              (keydown.enter)="onSort(col)"
              (keydown.space)="onSort(col)"
              [tabindex]="col.sortable ? 0 : -1"
            >
              <div class="th-content">
                <span>{{ col.label }}</span>
                @if (col.sortable) {
                  <span class="sort-indicator" aria-hidden="true">
                    @if (sortColumn === col.key) {
                      <ion-icon [name]="sortDirection === 'asc' ? 'arrow-up-outline' : 'arrow-down-outline'"></ion-icon>
                    } @else {
                      <ion-icon name="swap-vertical-outline" class="neutral-sort"></ion-icon>
                    }
                  </span>
                }
              </div>
            </th>
          }
          @if (showActions) {
            <th role="columnheader" scope="col" class="actions-header">Acciones</th>
          }
        </tr>
      </thead>
      <tbody>
        @if (!loading && paginatedData.length === 0) {
          <tr role="row">
            <td [attr.colspan]="columns.length + (showActions ? 1 : 0)" class="empty-cell">
              <div class="empty-state">
                <p>{{ emptyMessage }}</p>
              </div>
            </td>
          </tr>
        } @else {
          @for (item of paginatedData; track item.id || $index) {
            <tr
              role="row"
              class="table-row"
              (click)="onRowSelect(item)"
              (keydown.enter)="onRowSelect(item)"
              tabindex="0"
              aria-label="Fila de datos, presione enter para ver detalle"
            >
              @for (col of columns; track col.key) {
                <td role="cell" [attr.data-label]="col.label">
                  @if (col.type === 'image') {
                    <div
                      class="table-img-cell"
                      (click)="openImagePreview(item[col.key], item.name || item.sku, $event)"
                      [class.clickable]="!!item[col.key]"
                      [title]="item[col.key] ? 'Clic para ampliar imagen' : 'Sin imagen registrada'"
                    >
                      @if (item[col.key]) {
                        <img
                          [src]="item[col.key]"
                          [alt]="item.name || 'Miniatura'"
                          loading="lazy"
                          class="product-thumb-img"
                        />
                      } @else {
                        <div class="no-thumb-box">
                          <ion-icon name="image-outline"></ion-icon>
                        </div>
                      }
                    </div>
                  } @else if (col.type === 'badge') {
                    <ion-badge [color]="getBadgeColor(item[col.key])" class="status-badge">
                      {{ item[col.key] }}
                    </ion-badge>
                  } @else {
                    <span>{{ formatCellValue(item, col) }}</span>
                  }
                </td>
              }
              @if (showActions) {
                <td role="cell" class="actions-cell" (click)="$event.stopPropagation()">
                  <div class="actions-group">
                    @if (canView) {
                      <ion-button
                        fill="clear"
                        size="small"
                        color="secondary"
                        (click)="triggerAction('view', item, $event)"
                        [attr.aria-label]="'Ver detalle de ' + (item.name || item.sku || 'registro')"
                      >
                        <ion-icon slot="icon-only" name="eye-outline"></ion-icon>
                      </ion-button>
                    }
                    @if (canEdit) {
                      <ion-button
                        fill="clear"
                        size="small"
                        color="primary"
                        (click)="triggerAction('edit', item, $event)"
                        [attr.aria-label]="'Editar ' + (item.name || item.sku || 'registro')"
                      >
                        <ion-icon slot="icon-only" name="create-outline"></ion-icon>
                      </ion-button>
                    }
                    @if (canDelete) {

                        (click)="triggerAction('delete', item, $event)"

                      >
                        <ion-icon slot="icon-only" name="trash-outline"></ion-icon>
                      </ion-button>
                    }
                  </div>
                </td>
              }
            </tr>
          }
        }
      </tbody>
    </table>
  </div>

  <!-- Paginación Accesible -->
  @if (totalPages > 1) {
    <nav class="table-pagination" aria-label="Navegación por páginas de la tabla">

        Página <strong>{{ currentPage }}</strong> de <strong>{{ totalPages }}</strong>
      </div>
      <div class="pagination-buttons">

          size="small"

        >
          <ion-icon slot="icon-only" name="chevron-back-outline"></ion-icon>
        </ion-button>

        <span class="page-indicator" aria-current="page">{{ currentPage }}</span>

        <ion-button
          fill="outline"
          size="small"
          [disabled]="currentPage === totalPages"
          (click)="setPage(currentPage + 1)"
          aria-label="Ir a la página siguiente"
        >
          <ion-icon slot="icon-only" name="chevron-forward-outline"></ion-icon>
        </ion-button>
      </div>
    </nav>
  }

  <!-- Lightbox Modal de Previsualización de Imagen -->
  @if (previewModal(); as modal) {
    <div
      class="image-lightbox-overlay"
      (click)="closeImagePreview()"
      (keydown.escape)="closeImagePreview()"
      tabindex="0"
      role="dialog"
      aria-modal="true"
      aria-label="Vista previa ampliada de imagen"
    >
      <div class="image-lightbox-card" (click)="$event.stopPropagation()">
        <div class="lightbox-header">
          <span class="lightbox-title">{{ modal.title }}</span>
          <button type="button" class="lightbox-close-btn" (click)="closeImagePreview()" aria-label="Cerrar vista previa">
            <ion-icon name="close-outline"></ion-icon>
          </button>
        </div>
        <div class="lightbox-body">
          <img [src]="modal.url" [alt]="modal.title" class="lightbox-image" />
        </div>
      </div>
    </div>
  }
</div>
